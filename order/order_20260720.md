## 0.4 W5：跨目录 FlowerTune LOCO paired stats（0-GPU）

说明：该命令合并 `order_0717` 的 offset0 与 `order_0718` 的 offset1-4，生成干净统计 artifact。若服务器上目录命名不同，先用 `find /data2/minghao/result/FedPLoRA -path '*flowertune*offset*seed*/result_logs/*.json'` 确认。

```bash
nohup bash -lc '
set -euo pipefail
export BASE=/data2/minghao/result/FedPLoRA
export OUT=/data2/minghao/result/FedPLoRA/order_0720/audit_20260720
mkdir -p "$OUT/run_logs" "$OUT/analysis"
python - <<PY
import json
import math
import pathlib
import re
import statistics as stats

base = pathlib.Path("/data2/minghao/result/FedPLoRA")
out = pathlib.Path("/data2/minghao/result/FedPLoRA/order_0720/audit_20260720/analysis")
patterns = [
    "order_0717/**/flowertune*offset0*seed*/result_logs/*.json",
    "order_0717/**/flowertune*strict*heldout*seed*/result_logs/*.json",
    "order_0718/**/flowertune*offset*seed*/result_logs/*.json",
    "flowertune*offset0*seed*/result_logs/*.json",
    "flowertune*strict*heldout*seed*/result_logs/*.json",
    "flowertune*offset*seed*/result_logs/*.json",
]

def find_num(obj, names):
    if isinstance(obj, dict):
        for name in names:
            if name in obj and isinstance(obj[name], (int, float)) and math.isfinite(float(obj[name])):
                return float(obj[name])
        for value in obj.values():
            got = find_num(value, names)
            if got is not None:
                return got
    elif isinstance(obj, list):
        for value in obj:
            got = find_num(value, names)
            if got is not None:
                return got
    return None

def get_scheme_metrics(payload):
    results = payload.get("results", payload)
    if isinstance(results, dict) and "schemes" in results and isinstance(results["schemes"], dict):
        results = results["schemes"]
    schemes = {}
    if isinstance(results, dict):
        for name, item in results.items():
            if not isinstance(item, dict):
                continue
            macro = find_num(item, ["macro", "macro_acc", "domain_macro", "mean_acc", "avg_acc", "local_macro_acc"])
            worst = find_num(item, ["worst", "worst_acc", "worst_domain", "min_acc", "local_worst_acc"])
            if macro is not None:
                schemes[str(name)] = {"macro": macro, "worst": worst}
    return schemes

def infer_seed_offset(path, payload):
    text = str(path)
    seed = payload.get("seed") or payload.get("config", {}).get("seed")
    if seed is None:
        m = re.search(r"seed[_-]?(\\d+)", text)
        seed = int(m.group(1)) if m else None
    offset = payload.get("held_out_offset") or payload.get("config", {}).get("held_out_offset")
    if offset is None:
        m = re.search(r"offset[_-]?(\\d+)", text)
        offset = int(m.group(1)) if m else 0
    return int(seed), int(offset)

found = {}
for pat in patterns:
    for path in base.glob(pat):
        try:
            payload = json.loads(path.read_text(encoding="utf-8"))
        except Exception:
            continue
        schemes = get_scheme_metrics(payload)
        if not {"global", "coldstart"}.issubset(schemes):
            continue
        seed, offset = infer_seed_offset(path, payload)
        key = (offset, seed)
        # Prefer files with coldstart_geom/select_without_local when duplicates exist.
        score = len(schemes)
        prev = found.get(key)
        if prev is None or score > prev["score"]:
            found[key] = {"path": str(path), "score": score, "schemes": schemes}

expected = {(o, s) for o in range(5) for s in (42, 43, 44)}
missing = sorted(expected - set(found))
extra = sorted(set(found) - expected)
if missing:
    print("[W5][error] missing offset/seed units:", missing)
    print("[W5][debug] found:", sorted(found))
    raise SystemExit(2)
if extra:
    print("[W5][warn] extra offset/seed units ignored:", extra)

rows = []
for key in sorted(expected):
    entry = found[key]
    schemes = entry["schemes"]
    global_macro = schemes["global"]["macro"]
    global_worst = schemes["global"].get("worst")
    cold_macro = schemes["coldstart"]["macro"]
    cold_worst = schemes["coldstart"].get("worst")
    geom = schemes.get("coldstart_geom", schemes.get("select_without_local", schemes["coldstart"]))
    rows.append({
        "offset": key[0],
        "seed": key[1],
        "path": entry["path"],
        "global_macro": global_macro,
        "coldstart_macro": cold_macro,
        "coldstart_delta_macro": cold_macro - global_macro,
        "coldstart_geom_macro": geom["macro"],
        "coldstart_geom_delta_macro": geom["macro"] - global_macro,
        "global_worst": global_worst,
        "coldstart_worst": cold_worst,
        "coldstart_delta_worst": None if global_worst is None or cold_worst is None else cold_worst - global_worst,
    })

def summary(values):
    values = [float(v) for v in values if v is not None and math.isfinite(float(v))]
    n = len(values)
    mean = stats.mean(values)
    sd = stats.stdev(values) if n > 1 else 0.0
    ci = 1.96 * sd / math.sqrt(n) if n > 1 else 0.0
    return {"n": n, "mean": mean, "sd": sd, "ci95_low": mean - ci, "ci95_high": mean + ci}

result = {
    "units_expected": 15,
    "units_found": len(rows),
    "rows": rows,
    "summary": {
        "global_macro": summary([r["global_macro"] for r in rows]),
        "coldstart_macro": summary([r["coldstart_macro"] for r in rows]),
        "coldstart_delta_macro": summary([r["coldstart_delta_macro"] for r in rows]),
        "coldstart_geom_delta_macro": summary([r["coldstart_geom_delta_macro"] for r in rows]),
        "coldstart_delta_worst": summary([r["coldstart_delta_worst"] for r in rows]),
        "positive_macro_units": sum(1 for r in rows if r["coldstart_delta_macro"] > 0),
    },
}

out.mkdir(parents=True, exist_ok=True)
(out / "flower_loco_paired_stats_crossdir_20260720.json").write_text(json.dumps(result, indent=2, ensure_ascii=False), encoding="utf-8")
lines = [
    "# FlowerTune LOCO cross-directory paired stats 20260720",
    "",
    f"- units_found: {len(rows)}/15",
    f"- coldstart-global macro delta: {result['summary']['coldstart_delta_macro']['mean']:.4f} ± {result['summary']['coldstart_delta_macro']['sd']:.4f}",
    f"- approx 95% CI: [{result['summary']['coldstart_delta_macro']['ci95_low']:.4f}, {result['summary']['coldstart_delta_macro']['ci95_high']:.4f}]",
    f"- positive_macro_units: {result['summary']['positive_macro_units']}/15",
    "",
    "| offset | seed | global | coldstart | delta | source |",
    "|---:|---:|---:|---:|---:|---|",
]
for r in rows:
    lines.append(f"| {r['offset']} | {r['seed']} | {r['global_macro']:.4f} | {r['coldstart_macro']:.4f} | {r['coldstart_delta_macro']:.4f} | `{r['path']}` |")
(out / "flower_loco_paired_stats_crossdir_20260720.md").write_text("\\n".join(lines) + "\\n", encoding="utf-8")
print(json.dumps(result["summary"], indent=2, ensure_ascii=False))
PY
' > /data2/minghao/result/FedPLoRA/order_0720/audit_20260720/run_logs/test20260720_W5_flower_loco_crossdir_stats.log 2>&1 &
```

【串行与并行逻辑】

推荐执行顺序：

1. 先跑 `0.1-0.4`：0-GPU 前置、W6 fingerprint、W5 cross-dir stats。它们不占 GPU，可与 smoke 同时跑，但建议先跑，便于尽早发现路径问题。
2. 跑 `1.1-1.6 smoke`：smoke 结束后确认没有导入/数据/OOM 错误。
3. 跑 E1 route ablation：优先级最高。若只有 2 张 A100，建议：
   - 第一批并行：`2.1 + 2.2`；
   - 第二批并行：`2.3 + 2.4`；
   - 第三批并行：`2.5 + 2.6`；
   - 第四批并行：`2.7 + 2.8`。
4. 跑 E2 route probe：eval 级，通常比训练便宜；可穿插在 E1 空档跑。若资源紧张，优先 probe=1，其次 probe=2，最后 probe=5。
5. E3 Qwen3B 是条件分支：
   - 若 3B 不进正文：不要跑本节，保留 seed42 probe 并在 limitation 说明。
   - 若 3B 进正文：让合作者或空闲 GPU 单独跑本节，避免与 E1 抢资源。

优先级总表：

```text
必做且 0-GPU: W1-W6
必须优先 GPU: E1 route ablation seed43/44
建议补齐 GPU/eval: E2 FlowerTune probe seed43/44
条件 GPU: E3 Qwen3B seed43/44
明确不做: v14 fallback, μ/rank/A-grid sweep, CORE-8 3B 全分支, 第三数据集门槛化
```

【注意事项】

1. 本文件中所有正式训练命令都带 `--force_retrain`，避免命中旧 checkpoint 后直接 skip。
2. `coldstart_geom` 可写作 domain-label-free/few-shot geometric onboarding，但不要写 zero-data。
3. Qwen3B 命令故意使用显式 `tasks/fed_train_sft.py`，避免此前 main/baseline preflight guard 混用导致误拦截。
4. 运行前确认当前目录是 `/data2/minghao/code/FedPLoRA-main`，否则相对脚本路径会找不到。
